## Usage

```tsx
import { Checkbox } from "@/components/ui/checkbox"
import { Field } "@/components/ui/field"
```

```tsx
<Field>
  <Checkbox />
  <Field.Label>
    Accept terms and conditions
  </Field.Label>
</Field>
```
