## Usage

```tsx showLineNumbers
import { Field } from "@/components/ui/field"
import { Radio } from "@/components/ui/radio"
```

```tsx showLineNumbers
<Radio.Group defaultValue="option-one">
  <Field className="flex items-center gap-3">
    <Radio value="option-one" id="option-one" />
    <Field.Label htmlFor="option-one">Option One</Field.Label>
  </Field>
  <Field className="flex items-center gap-3">
    <Radio value="option-two" id="option-two" />
    <Field.Label htmlFor="option-two">Option Two</Field.Label>
  </Field>
</Radio.Group>
```

## Composition

Use the following composition to build a `Radio`:

```text
Radio.Group
├── Radio
└── Radio
```

or prefer RadioGroup

```text
RadioGroup
├── RadioGroup.Item
└── RadioGroup.Item
```
