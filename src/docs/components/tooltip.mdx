## Installation
<Install component='tooltip' />

<Steps className='mt-4'>
<Step title={"Add the `TooltipProvider` to the root of your app."}>


```tsx title="app/layout.tsx" showLineNumbers {1,7}
import { TooltipProvider } from "@/components/ui/tooltip"

export default function RootLayout({ children }) {
  return (
    <html lang="en">
      <body>
        <TooltipProvider>{children}</TooltipProvider>
      </body>
    </html>
  )
}
```
</Step>

</Steps>

## Usage

```tsx showLineNumbers
import { Tooltip } from "@/components/ui/tooltip"
```

```tsx showLineNumbers
<Tooltip>
  <Tooltip.Trigger>Hover</Tooltip.Trigger>
  <Tooltip.Content>
    <p>Add to library</p>
  </Tooltip.Content>
</Tooltip>
```

## Composition

Use the following composition to build a `Tooltip`:

```text
Tooltip
├── Tooltip.Trigger
└── Tooltip.Content
```
