## Usage

```tsx showLineNumbers
import { OTPField } from "@/components/ui/input-otp"
```

```tsx showLineNumbers
<OTPField length={6}>
  <OTPField.Input />
  <OTPField.Input />
  <OTPField.Input />
  <OTPField.Separator />
  <OTPField.Input />
  <OTPField.Input />
  <OTPField.Input />
</OTPField>
```

## Composition

Use the following composition to build an `OTPField`:

```text
OTPField
├── OTPField.Input
├── OTPField.Input
├── OTPField.Input
├── OTPField.Separator
├── OTPField.Input
├── OTPField.Input
└── OTPField.Input
```

## Examples

### Pattern

Use the `validationType` prop to define a custom pattern for the OTP input.

```tsx showLineNumbers {1,5}
<OTPField length={6} validationType="numeric">
  ...
</OTPField>
```
