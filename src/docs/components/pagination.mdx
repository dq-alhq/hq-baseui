## Usage

```tsx showLineNumbers
import { Pagination } from "@/components/ui/pagination"
```

```tsx showLineNumbers
<Pagination>
  <Pagination.First href="#" />
  <Pagination.Prev href="#" />
  <Pagination.Item href="#">1</Pagination.Item>
  <Pagination.Item href="#" isActive>
    2
  </Pagination.Item>
  <Pagination.Item href="#">3</Pagination.Item>
  <Pagination.Ellipsis />
  <Pagination.Next href="#" />
  <Pagination.Last href="#" />
</Pagination>
```

## Composition

Use the following composition to build a `Pagination`:

```text
Pagination
├── Pagination.First
├── Pagination.Prev
├── Pagination.Item
├── Pagination.Ellipsis
└── Pagination.Next
└── Pagination.Last
```

## API References

### Pagination

| Prop      | Type                           | Default     |
| --------- | ------------------------------ | ----------- |
| `variant` | `"outline" \| "ghost"`         | `"outline"` |
| `size`    | `"xs" \| "sm" \| "md" \| "lg"` | `"md"`      |
| `spacing` | `number`                       | 0           |
