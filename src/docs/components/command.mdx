## Usage

```tsx showLineNumbers
import { Command } from "@/components/ui/command"
```

```tsx showLineNumbers
<Command.Dialog items={groups} onOpenChange={setOpen} open={open}>
  <CommandInput placeholder="Search..." />
  <Command.Empty>No matches found.</Command.Empty>
  <Command.List>
    {(group) => (
      <Fragment key={group.id}>
        <Command.Group heading={group.heading}>
          {group.items.map((item) => (
            <Command.Item key={item.id} value={item.value}>
              {item.label}
            </Command.Item>
          ))}
        </Command.Group>
        <Command.Separator />
      </Fragment>
    )}
  </Command.List>
  <Command.Footer>
    Footer content
  </Command.Footer>
</Command.Dialog>
```

## Composition

```text
Command or Command.Dialog
├── Command.Input
├── Command.Empty
├── Command.List
│   ├── Command.Group
│   │   └── Command.Item
│   └── Command.Separator
└── Command.Footer
```
